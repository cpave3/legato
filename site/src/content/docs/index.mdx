import { Card, CardGrid, LinkCard } from '@astrojs/starlight/components';

<CardGrid stagger>
  <Card title="A board in your terminal" icon="seti:todo">
    A kanban board you drive with the keyboard. Keep local tasks, or pull tickets from Jira and push your moves back.
  </Card>
  <Card title="Agents on every task" icon="rocket">
    Start Claude Code, Codex or Chimera on a task. Each card shows whether its agent is running, waiting for you, or idle.
  </Card>
  <Card title="Web UI and phone" icon="laptop">
    The same board and agents in the browser and on your phone. Talk to an agent, answer its questions and approve its requests away from your desk.
  </Card>
  <Card title="Attention" icon="warning">
    One queue of everything that needs you: a plan to approve, a review to read, a failed workflow, a PR with a red check.
  </Card>
  <Card title="Plans and reviews" icon="document">
    Agents propose a plan before they write code, and walk you through the change afterwards. You comment, ask and approve in Legato.
  </Card>
  <Card title="Credentials and sandbox" icon="seti:lock">
    Agents use your SSH keys and API keys without holding them, with each use approved by you. A sandbox hides the rest of your machine.
  </Card>
</CardGrid>
